$by = function (PDO $p, string $t, string $col) use ($has, $rows): string {   // "value:count,…" of one column of a P6 table, or "-"
    if (!$has($p, $t)) return 'absent';
    $o = [];
    foreach ($rows($p, 'SELECT ' . $col . ' AS v, COUNT(*) AS n FROM ' . $t . ' GROUP BY ' . $col . ' ORDER BY ' . $col) as $r) {
        $o[] = (string)preg_replace('/[^a-z0-9_]/', '', (string)($r['v'] ?? 'null')) . ':' . (int)$r['n'];
    }
    return $o === [] ? '-' : implode(',', $o);
};

// The census OF-G2 compares: every table of DB but the runner's own ledger (_migrations, _schema_checks — what the trial
// writes by design, and the verdict QF-G2 reads) and SQLite's own (sqlite_%): its columns as they are now, its row count
// and a hash of every row over those columns, in rowid order (or every column's, for a table without one). Values are
// typed and length-prefixed, so a BLOB or a NUL byte hashes as it is.
$census = function (PDO $p, ?array $cols = null) use ($has): array {
    $out = [];
    $names = $cols !== null ? array_keys($cols)
           : $p->query("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' AND name NOT IN ('_migrations', '_schema_checks') ORDER BY name")->fetchAll(PDO::FETCH_COLUMN);
    foreach ($names as $t) {
        $t = (string)$t;
        if (!$has($p, $t)) { $out[$t] = ['cols' => $cols[$t] ?? [], 'n' => -1, 'h' => 'absent']; continue; }
        $c = $cols !== null ? $cols[$t] : array_map(function ($r) { return (string)$r['name']; }, $p->query('PRAGMA table_info("' . str_replace('"', '""', $t) . '")')->fetchAll(PDO::FETCH_ASSOC));
        $sel = implode(',', array_map(function ($x) { return '"' . str_replace('"', '""', $x) . '"'; }, $c));
        try {
            try { $st = $p->query('SELECT ' . $sel . ' FROM "' . str_replace('"', '""', $t) . '" ORDER BY rowid'); }
            catch (Throwable $e) { $st = $p->query('SELECT ' . $sel . ' FROM "' . str_replace('"', '""', $t) . '" ORDER BY ' . implode(',', range(1, max(1, count($c))))); }
            $h = hash_init('sha256'); $n = 0;
            while (($r = $st->fetch(PDO::FETCH_NUM)) !== false) {
                $n++;
                foreach ($r as $v) {
                    if ($v === null) hash_update($h, 'N;');
                    elseif (is_int($v)) hash_update($h, 'i' . $v . ';');
                    elseif (is_float($v)) hash_update($h, 'f' . var_export($v, true) . ';');
                    else { $v = (string)$v; hash_update($h, 's' . strlen($v) . ':' . $v . ';'); }
                }
                hash_update($h, "\n");
            }
            $st->closeCursor();
            $out[$t] = ['cols' => $c, 'n' => $n, 'h' => substr(hash_final($h), 0, 32)];
        } catch (Throwable $e) {
            $out[$t] = ['cols' => $c, 'n' => -2, 'h' => 'unreadable'];   // never compared as unchanged: OF-G2 says it and refuses
        }
    }
    return $out;
};

                     // by table and state, never FAIL as history), and what was added since OF-CA3's redeploy baseline
                     // (BASE: "max=<4 ids> ledger_max=<id> seen_max=<id>", or empty on a first deploy), each row by its own
                     // stamp against the copy's start, in a later second ("late") or not
        [$db, $base, $cu] = $a + ['', '', '0'];
        $p = $ro($db); $cu = (int)$cu;
        $bm = []; $lm = 0; $sm = 0;
        if (preg_match('/(?:^| )max=([0-9,]+)/', (string)$base, $m)) $bm = array_map('intval', explode(',', $m[1]));
        if (preg_match('/(?:^| )ledger_max=(\d+)/', (string)$base, $m)) $lm = (int)$m[1];
        if (preg_match('/(?:^| )seen_max=(\d+)/', (string)$base, $m)) $sm = (int)$m[1];
        $bm = $bm + [0, 0, 0, 0];
        $n = []; foreach ($P6 as $t) $n[] = $has($p, $t) ? (int)$one($p, 'SELECT COUNT(*) FROM ' . $t) : -1;
        $late = function (string $col): string { return "CAST(strftime('%s', " . $col . ") AS INTEGER) > :cu"; };
        $L = $has($p, 'wa_auto_sends'); $S = $has($p, 'evo_webhook_seen');
        $oAll = $L ? (int)$one($p, "SELECT COUNT(*) FROM wa_auto_sends WHERE kind = 'offer' OR ref GLOB 'offer_test:*'") : -1;
        $eAll = $S ? (int)$one($p, "SELECT COUNT(*) FROM evo_webhook_seen WHERE event GLOB 'offer.*'") : -1;
        // FAIL: an added offer claim, an added offer.send echo or an added test, each stamped in a later second than the
        // copy's start; an added offer_test:<id> ledger row that names no test, or a test so started
        $f1 = $L ? (int)$one($p, "SELECT COUNT(*) FROM wa_auto_sends WHERE id > :m AND kind = 'offer' AND " . $late('claimed_at'), [':m' => $lm, ':cu' => $cu]) : 0;
        $f2 = $S ? (int)$one($p, "SELECT COUNT(*) FROM evo_webhook_seen WHERE id > :m AND event = 'offer.send' AND " . $late('received_at'), [':m' => $sm, ':cu' => $cu]) : 0;
        $f3 = $has($p, 'wa_offer_tests') ? (int)$one($p, "SELECT COUNT(*) FROM wa_offer_tests WHERE id > :m AND " . $late('started_at'), [':m' => $bm[2], ':cu' => $cu]) : 0;
        $f4 = 0;
        if ($L) {
            foreach ($rows($p, "SELECT ref FROM wa_auto_sends WHERE id > :m AND ref GLOB 'offer_test:*'", [':m' => $lm]) as $r) {
                $tid = substr((string)$r['ref'], 11);
                if (!preg_match('/^\d+$/', $tid) || !$has($p, 'wa_offer_tests')) { $f4++; continue; }
                $s = $one($p, "SELECT CAST(strftime('%s', started_at) AS INTEGER) FROM wa_offer_tests WHERE id = ?", [(int)$tid]);
                if ($s === false || $s === null || (int)$s > $cu) $f4++;
            }
        }
        // counted and printed: what a writer that decided before the clear may have added (ruling R18; §H.4.21)
        $c1 = $L ? (int)$one($p, "SELECT COUNT(*) FROM wa_auto_sends WHERE id > :m AND kind = 'offer'", [':m' => $lm]) - $f1 : 0;
        $c2 = $has($p, 'wa_offer_tests') ? (int)$one($p, 'SELECT COUNT(*) FROM wa_offer_tests WHERE id > :m', [':m' => $bm[2]]) - $f3 : 0;
        $c3 = $L ? (int)$one($p, "SELECT COUNT(*) FROM wa_auto_sends WHERE id > :m AND ref GLOB 'offer_test:*'", [':m' => $lm]) - $f4 : 0;
        $c4 = $S ? (int)$one($p, "SELECT COUNT(*) FROM evo_webhook_seen WHERE id > :m AND event GLOB 'offer.*'", [':m' => $sm]) - $f2 : 0;
        $ad = [];
        if ($has($p, 'wa_offers')) $ad[] = 'offers:' . (int)$one($p, 'SELECT COUNT(*) FROM wa_offers WHERE id > ?', [$bm[0]]);
        if ($has($p, 'wa_offer_recipients')) $ad[] = 'recipients:' . (int)$one($p, 'SELECT COUNT(*) FROM wa_offer_recipients WHERE id > ?', [$bm[1]]);
        $ev = [];
        if ($has($p, 'wa_offer_events')) foreach ($rows($p, 'SELECT action, COUNT(*) AS n FROM wa_offer_events WHERE id > ? GROUP BY action ORDER BY action', [$bm[3]]) as $r) $ev[] = preg_replace('/[^a-z_]/', '', (string)$r['action']) . ':' . (int)$r['n'];
        echo 'rows=', implode(',', $n), ' ledger_offer=', $oAll, ' seen_offer=', $eAll,
             ' late_claims=', $f1, ' late_echoes=', $f2, ' late_tests=', $f3, ' bad_test_rows=', $f4,
             ' added_claims=', $c1, ' added_tests=', $c2, ' added_test_rows=', $c3, ' added_echoes=', $c4,
             ' added=', $csv($ad), ' added_events=', $csv($ev),
             ' offers=', $by($p, 'wa_offers', 'state'), ' recipients=', $by($p, 'wa_offer_recipients', 'state'), ' tests=', $by($p, 'wa_offer_tests', 'state');
        break;

    case 'ca4':      // ca4 ROOT DIR — OF-CA4's in-memory half through ROOT's (the installed) OfferSwitch: DIR's copy's own
                     // configuration with both offer keys on and offers_live_since now, forced South Sudan (live() and
                     // screen() false: the gate, not the switches) and on Uganda (the control: both true)
        [$root, $dir] = $a;
        $lib($root, ['bootstrap_data', 'timezone', 'UtcClock', 'PluginConfig', 'TenantProfile', 'StaffJobsGate', 'OfferSwitch']);
        $c = PluginConfig::read($root, $dir);
        $c['offers_test_enabled'] = '1'; $c['offers_live_enabled'] = '1'; $c['offers_live_since'] = gmdate('Y-m-d H:i:s');
        $ss = $c; $ss['tenant_profile'] = 'south-sudan';
        $ug = $c; $ug['tenant_profile'] = 'uganda';
        echo 'ss_live=', OfferSwitch::live($ss, $dir) ? 1 : 0, ' ss_screen=', OfferSwitch::screen($ss, $dir) ? 1 : 0,
             ' ug_live=', OfferSwitch::live($ug, $dir) ? 1 : 0, ' ug_screen=', OfferSwitch::screen($ug, $dir) ? 1 : 0;
        break;

    case 'prep4':    // prep4 SRC DST — OF-CA4's scratch settings folder DST beside SRC's copy: SRC's settings copies with the
                     // profile forced to South Sudan and both offer keys on (offers_live_since now), its vault's tenant
                     // forced the same, and NO book: the job and the tool may open none (a book appearing is the fault)
        [$src, $dst] = $a;
        if (!$mine($src) || !$mine($dst) || !is_dir($dst)) { echo '?notacopy'; exit(2); }
        $k = is_file($src . '/kyc_config.json') ? json_decode((string)file_get_contents($src . '/kyc_config.json'), true) : [];
        if (!is_array($k)) $k = [];
        $k['tenant_profile'] = 'south-sudan'; $k['offers_test_enabled'] = '1'; $k['offers_live_enabled'] = '1'; $k['offers_live_since'] = gmdate('Y-m-d H:i:s');
        if (!$put($dst . '/kyc_config.json', json_encode($k, JSON_UNESCAPED_SLASHES | JSON_FORCE_OBJECT))) { echo '?write'; exit(3); }
        if (is_file($src . '/config.json') && !@copy($src . '/config.json', $dst . '/config.json')) { echo '?write'; exit(3); }
        $v = is_file($src . '/vault.json') ? json_decode((string)file_get_contents($src . '/vault.json'), true) : null;
        if (is_array($v)) {
            if (!isset($v['config']) || !is_array($v['config'])) $v['config'] = [];
            $v['config']['tenant_profile'] = 'south-sudan';
            if (!$put($dst . '/vault.json', json_encode($v, JSON_UNESCAPED_SLASHES))) { echo '?write'; exit(3); }
        }
        echo 'prep=ok book=', is_file($dst . '/plugin.sqlite3') ? 'present' : 'none', ' vault=', is_array($v) ? 'forced' : 'none';
        break;

    case 'job':      // job ROOT DIR — OF-CA4's job: ROOT's (the installed) cron/offers_send.php included directly, as master
                     // includes it, in this process alone, on the scratch DIR (prep4's): the data folder it resolves must be
                     // DIR and live() on DIR's settings must read false, or it is never included; then nothing written (no
                     // book appears in DIR), no sender, audience or gateway class loaded, nothing printed — and the network
                     // is off in this process whatever it would do
        [$root, $dir] = $a;
        if (!$mine($dir) || is_file($dir . '/plugin.sqlite3')) { echo '?notascratch'; exit(2); }
        $lib($root, ['bootstrap_data', 'timezone', 'UtcClock', 'PluginConfig', 'TenantProfile', 'StaffJobsGate', 'OfferSwitch']);
        if (rtrim(getDataDir($root), '/') !== rtrim($dir, '/')) { echo 'refused=datadir'; break; }
        if (OfferSwitch::live(PluginConfig::read($root, $dir), $dir)) { echo 'refused=live_true'; break; }
        $net = (string)ini_get('allow_url_fopen') === '0' && strpos((string)ini_get('disable_functions'), 'curl_exec') !== false ? 'off' : 'on';
        if ($net !== 'off') { echo 'refused=network_on'; break; }
        $_m_deadline = time() + 30;
        ob_start();
        try { include $root . '/cron/offers_send.php'; $thrown = '-'; } catch (Throwable $e) { $thrown = get_class($e); }
        $out = (string)ob_get_clean();
        echo 'datadir=scratch pre_live=0 net=off thrown=', $tok($thrown), ' out_bytes=', strlen($out),
             ' book=', is_file($dir . '/plugin.sqlite3') ? 'created' : 'none',
             ' wal=', (is_file($dir . '/plugin.sqlite3-wal') || is_file($dir . '/plugin.sqlite3-shm')) ? 'present' : 'none',
             ' sender=', class_exists('OfferSender', false) ? 1 : 0, ' audience=', class_exists('OfferAudience', false) ? 1 : 0,
             ' gateway=', class_exists('EvolutionOfferGateway', false) ? 1 : 0;
        break;

    case 'ca5':      // ca5 ROOT — OF-CA5 (S2-11): ROOT's (the installed) OfferSwitch::declaredKeys() against the union of its
                     // SalesFollowupLadder::allKeys() and settingKeys() (ruling R5)
        $lib($a[0], ['UtcClock', 'SalesFollowupLadder', 'OfferSwitch']);
        $d = OfferSwitch::declaredKeys();
        $u = array_merge(SalesFollowupLadder::allKeys(), SalesFollowupLadder::settingKeys());
        $out = array_values(array_diff($d, $u));
        echo 'declared=', count($d), ' all=', count(SalesFollowupLadder::allKeys()), ' settings=', count(SalesFollowupLadder::settingKeys()), ' outside=', $csv($out);
        break;

    case 'rb':       // rb DB IDS START — OF-RB1 on a copy taken after QF-RB1, before the put-back: Part O's end state (no
                     // offer running, no recipient claimed), what its hook left (offers paused by the rollback and their
                     // events, recipients settled uncertain as stale claims, returned to queued, tests left sending); the
                     // paused offers' ids to IDS (the sheet). START is QF-RB1's start (UTC seconds, taken before the tool
                     // ran): the hook makes uncertain every test sending since before ITS moment less STALE_S, and that
                     // moment is later than START, so a test still sending that had started before START less STALE_S is
                     // one the hook should have settled (tests_stale); one that started after is one it rightly left,
                     // under 15 minutes old when it ran (tests_young) — never "now" less 15 minutes: the copy is taken
                     // after the hooks, the other parts' steps and the read-backs (ruling R18)
        [$db, $idf, $st0] = array_pad($a, 3, '');
        if (!$mine($idf)) { echo '?notacopy'; exit(2); }
        if (!preg_match('/^\d{9,11}$/', (string)$st0)) { echo '?start'; exit(2); }
        $cut = gmdate('Y-m-d H:i:s', (int)$st0 - @@OF_STALE_S@@);   // UtcClock::stamp()'s form, as started_at is written
        $p = $ro($db);
        if (!$has($p, 'wa_offers')) { echo 'tables=absent'; break; }
        $run = (int)$one($p, "SELECT COUNT(*) FROM wa_offers WHERE state = 'running'");
        $cl = (int)$one($p, "SELECT COUNT(*) FROM wa_offer_recipients WHERE state = 'claimed'");
        $sc = (int)$one($p, "SELECT COUNT(*) FROM wa_offer_recipients WHERE state = 'uncertain' AND reason = 'stale_claim'");
        $stale = (int)$one($p, "SELECT COUNT(*) FROM wa_offer_tests WHERE state = 'sending' AND started_at < ?", [$cut]);
        $young = (int)$one($p, "SELECT COUNT(*) FROM wa_offer_tests WHERE state = 'sending' AND started_at >= ?", [$cut]);
        $pe = (int)$one($p, "SELECT COUNT(*) FROM wa_offer_events WHERE action = 'paused_by_rollback'");
        $ids = array_map('intval', $p->query("SELECT id FROM wa_offers WHERE state = 'paused' ORDER BY id")->fetchAll(PDO::FETCH_COLUMN));
        if (!$put($idf, ($ids === [] ? 'paused offers: none' : 'paused offers: offer ' . implode(', offer ', array_slice($ids, 0, 200))) . "\n")) { echo '?write'; exit(3); }
        echo 'tables=present running=', $run, ' claimed=', $cl, ' stale_claim=', $sc, ' tests_stale=', $stale, ' tests_young=', $young,
             ' paused_by_rollback=', $pe, ' offers=', $by($p, 'wa_offers', 'state'), ' recipients=', $by($p, 'wa_offer_recipients', 'state');
        break;

    case 'rb2':      // rb2 DB — OF-RB2 on a copy after the put-back: the four tables and 091's _schema_checks row stay; the
                     // offers by state (a paused one resumes only by an admin's Resume, after a redeploy)
        $p = $ro($a[0]);
        $k = 0; foreach ($P6 as $t) if ($has($p, $t)) $k++;
        $sc = ($has($p, '_schema_checks') && (int)$one($p, 'SELECT COUNT(*) FROM _schema_checks WHERE migration = ? AND found = expected', ['091_wa_offers.sql']) > 0) ? 'present' : 'absent';
        echo 'tables=', $k, '/', count($P6), ' sc=', $sc, ' offers=', $by($p, 'wa_offers', 'state'),
             ' running=', $has($p, 'wa_offers') ? (int)$one($p, "SELECT COUNT(*) FROM wa_offers WHERE state = 'running'") : 0,
             ' paused=', $has($p, 'wa_offers') ? (int)$one($p, "SELECT COUNT(*) FROM wa_offers WHERE state = 'paused'") : 0;
        break;

    default:
        echo '?mode'; exit(2);
    }
} catch (Throwable $e) { echo '?', get_class($e); exit(3); }
