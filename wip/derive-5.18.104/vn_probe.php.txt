$ro = function (string $db) {
    if (!is_file($db)) throw new RuntimeException('nodb');
    $p = new PDO('sqlite:' . $db, null, null, [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_TIMEOUT => 15,
                                                PDO::SQLITE_ATTR_OPEN_FLAGS => PDO::SQLITE_OPEN_READONLY]);
    $p->exec('PRAGMA query_only = ON');
    return $p;
};
$has = function (PDO $p, string $t): bool {
    $s = $p->prepare("SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = ?"); $s->execute([$t]);
    return $s->fetchColumn() !== false;
};
$one = function (PDO $p, string $sql, array $x = []) { $s = $p->prepare($sql); $s->execute($x); return $s->fetchColumn(); };
$rows = function (PDO $p, string $sql, array $x = []): array { $s = $p->prepare($sql); $s->execute($x); return $s->fetchAll(PDO::FETCH_ASSOC); };
$tok = function ($s): string { return (string)preg_replace('/[^A-Za-z0-9_.:\/+-]/', '', (string)$s); };
$setNow = function (array $c, string $k): bool { return array_key_exists($k, $c) && $c[$k] !== null && $c[$k] !== '' && $c[$k] !== false; };
$flag = function ($v): bool { if (is_bool($v)) return $v; return in_array(strtolower(trim((string)$v)), ['1', 'true', 'on', 'yes'], true); };
$lib = function (string $root, array $names) { foreach ($names as $l) require_once rtrim($root, '/') . '/lib/' . $l . '.php'; };
$csv = function (array $x): string { return $x === [] ? '-' : implode(',', $x); };
$pct = function (array $v, int $q): int {
    if ($v === []) return 0;
    sort($v);
    $i = (int)ceil($q / 100 * count($v)) - 1;
    return (int)$v[max(0, min(count($v) - 1, $i))];
};

// Part V's switch and record keys (its five rows' rowKeys and their sf_skip_/sf_override_ keys, as allKeys() has them)
// and its settings (settingKeys() whose part is V), from a code root's SalesFollowupLadder
$vkeys = function (string $root) use ($lib, $VSTEPS): array {
    $lib($root, ['UtcClock', 'SalesFollowupLadder']);
    $all = SalesFollowupLadder::allKeys(); $sw = [];
    foreach ($VSTEPS as $s) {
        if (SalesFollowupLadder::row($s) === null) throw new RuntimeException('step');
        foreach (SalesFollowupLadder::rowKeys($s) as $k) $sw[] = $k;
        $sw[] = 'sf_skip_' . $s; $sw[] = 'sf_override_' . $s;
    }
    $sw = array_values(array_intersect(array_values(array_unique($sw)), $all));
    $st = [];
    foreach (SalesFollowupLadder::SETTINGS as $k => $d) {
        if ((string)($d['part'] ?? '') === 'V' && in_array($k, SalesFollowupLadder::settingKeys(), true)) $st[] = $k;
    }
    return [$sw, $st];
};
