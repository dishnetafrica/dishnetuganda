$ro = function (string $db) {
    $p = new PDO('sqlite:' . $db, null, null, [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_TIMEOUT => 15,
                                                PDO::SQLITE_ATTR_OPEN_FLAGS => PDO::SQLITE_OPEN_READONLY]);
    $p->exec('PRAGMA query_only = ON');
    return $p;
};
$rw = function (string $db) {
    $p = new PDO('sqlite:' . $db, null, null, [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION, PDO::ATTR_TIMEOUT => 15]);
    $p->exec('PRAGMA busy_timeout = 5000');
    return $p;
};
$has = function (PDO $p, string $t): bool {
    $s = $p->prepare("SELECT 1 FROM sqlite_master WHERE type = 'table' AND name = ?"); $s->execute([$t]);
    return $s->fetchColumn() !== false;
};
$col = function (PDO $p, string $t, string $c) use ($has): bool {
    if (!$has($p, $t)) return false;
    foreach ($p->query('PRAGMA table_info(' . $t . ')')->fetchAll(PDO::FETCH_ASSOC) as $r) if ((string)$r['name'] === $c) return true;
    return false;
};

// a SqliteStore JSON document, read as SqliteStore::load() reads it (lib/SqliteStore.php:339): the table is the file's
// name; a config document (master_schedule.json) is one row with the sentinel id 0, a record list (retailers.json) a row each
$flat = function (PDO $p, string $name) use ($has) {
    $t = preg_replace('/[^a-zA-Z0-9_]/', '_', preg_replace('/\.json$/', '', $name));
    if (!$has($p, $t)) return null;
    $j = $p->query('SELECT data FROM [' . $t . '] WHERE id = 0 LIMIT 1')->fetchColumn();
    $d = is_string($j) ? json_decode($j, true) : null;
    return is_array($d) ? $d : null;
};
$docRows = function (PDO $p, string $name) use ($has) {
    $t = preg_replace('/[^a-zA-Z0-9_]/', '_', preg_replace('/\.json$/', '', $name));
    if (!$has($p, $t)) return null;
    $out = [];
    foreach ($p->query('SELECT data FROM [' . $t . '] ORDER BY id ASC')->fetchAll(PDO::FETCH_COLUMN) as $j) { $d = json_decode((string)$j, true); if (is_array($d)) $out[] = $d; }
    return $out;
};

// not JSON); and how many entries it holds besides the four jobs' — the positive control that the schedule was read at
// all, without which "no entry for the four" would pass over a schedule nobody read (QF-CA3, the redeploy baseline)
$sched = function (PDO $p) use ($has, $JOBS): array {
    if (!$has($p, 'master_schedule')) return ['absent', 0];
    $j = $p->query('SELECT data FROM [master_schedule] WHERE id = 0 LIMIT 1')->fetchColumn();
    if (!is_string($j)) return ['absent', 0];
    $d = json_decode($j, true);
    if (!is_array($d)) return ['unreadable', 0];
    $n = 0; foreach ($d as $k => $e) if (is_array($e) && !in_array((string)$k, $JOBS, true)) $n++;
    return ['read', $n];
};
