$cutEs = function (string $src, string $dst, array $keys) use ($put): array {
    if (!is_file($src)) return ['absent', 0];
    $raw = is_readable($src) ? @file_get_contents($src) : false;              // as QuoteAgentCopy::mailSetting reads it
    if ($raw === false) return ['unreadable', 0];
    $d = json_decode((string)$raw, true);
    $c = [];
    if (is_array($d)) foreach ($keys as $k) if (array_key_exists($k, $d)) $c[$k] = $d[$k];
    if (!$put($dst, json_encode($c, JSON_UNESCAPED_SLASHES | JSON_FORCE_OBJECT))) return ['write', 0];
    return ['present', count($c)];
};

        }
        echo implode(' ', $out), ' making=', $mk;
        break;

    default:
        echo '?mode'; exit(2);
    }
} catch (Throwable $e) { echo '?', get_class($e); exit(3); }
